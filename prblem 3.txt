class Solution:
    def lengthOfLongestSubstring(self, s):
        char_index = {}  # Dictionary to store the last seen index of each character
        max_length = 0
        left = 0         # Left pointer of the sliding window
        
        for right in range(len(s)):
            current_char = s[right]
            
            # If the character is in our map and inside our current window
            if current_char in char_index and char_index[current_char] >= left:
                left = char_index[current_char] + 1
            
            char_index[current_char] = right
            max_length = max(max_length, right - left + 1)
            
        return max_length
